String buildExtractPageTextScript() {
  return r'''
(function() {
  const elements = document.querySelectorAll('label, span, div, p, h1, h2, h3, h4, h5, h6, option');
  const texts = [];
  const seen = new Set();
  elements.forEach(el => {
    const text = el.textContent.trim();
    if (text && text.length > 2 && text.length < 200 && !seen.has(text)) {
      seen.add(text);
      texts.push(text);
    }
  });
  return JSON.stringify(texts.slice(0, 50));
})();
''';
}

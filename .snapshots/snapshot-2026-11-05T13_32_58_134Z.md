                'Cmd-F':  function () { editorSearchOpen(); }
            }
        });

        console.log('CodeMirror ready');
    } else {
        console.warn('CodeMirror not available, falling back to textarea');

function getEditorValue() {
    return window.cm ? window.cm.getValue()
                     : (document.getElementById('code-editor') || {value:''}).value;
}

function setEditorValue(text) {
    if (window.cm) {
        window.cm.getDoc().setValue(text);
        window.cm.refresh();
    } else {
        var el = document.getElementById('code-editor');
        if (el) el.value = text;
        if (window.aceEditorRefresh) window.aceEditorRefresh();
    }
}

function clearEditor() {
    setEditorValue('');
}

function copyEditor() {
    var text = getEditorValue();
    if (navigator.clipboard) {
        navigator.clipboard.writeText(text).then(function () { addOutput('Copied to clipboard.'); });
    } else {
        var ta = document.createElement('textarea');
        ta.value = text;
        document.body.appendChild(ta);
        ta.select();
        document.execCommand('copy');
        document.body.removeChild(ta);
        addOutput('Copied to clipboard.');
    }
}

function saveCurrentFile() {
    if (!window.currentFile) { alert('No file open.'); return; }
    var content = getEditorValue();
    fetch('/api/save-file', {
        method:  'POST',
        headers: { 'Content-Type': 'application/json' },
        body:    JSON.stringify({ filename: window.currentFile, content: content })
    })
    .then(function (r) { return r.json(); })
    .then(function (d) {
        addOutput(d.ok ? '✅ Saved: ' + window.currentFile
                       : '❌ Save failed: ' + (d.error || 'Unknown'));
    })
    .catch(function (e) { addOutput('❌ Save error: ' + e.message); });
}

// EXPLORER  — extracted to window.loadExplorer so the

window.currentFile = null;

window.loadExplorer = function () {
    var explorerDiv = document.getElementById('explorer');
    if (!explorerDiv) return;
    explorerDiv.innerHTML = 'Loading...';

    fetch('/api/files')
        .then(function (response) {
            if (!response.ok) throw new Error('HTTP ' + response.status);
            return response.json();
        })
        .then(function (data) {
            console.log('Explorer: received', data.length, 'paths');
            explorerDiv.innerHTML = '';

            // Build nested tree from flat paths
            var tree = {};
            data.forEach(function (path) {
                var parts = path.split('/');
                var current = tree;
                parts.forEach(function (part, index) {
                    if (index === parts.length - 1) {
                        current[part] = 'file:' + path;
                    } else {
                        if (!current[part] || typeof current[part] === 'string') {
                            current[part] = {};
                        }
                        current = current[part];
                    }
                });
            });

            function renderTree(node, parentElement, indentLevel) {
                var keys = Object.keys(node).sort(function (a, b) {
                    var aIsFile = typeof node[a] === 'string';
                    var bIsFile = typeof node[b] === 'string';
                    if (!aIsFile && bIsFile) return -1;
                    if (aIsFile && !bIsFile) return 1;
                    return a.localeCompare(b);
                });

                keys.forEach(function (key) {
                    var value = node[key];
                    var div = document.createElement('div');
                    div.style.padding      = '3px';
                    div.style.paddingLeft  = (indentLevel * 12) + 'px';
                    div.style.cursor       = 'pointer';
                    div.style.color        = '#ff9d00';
                    div.style.fontFamily   = 'monospace';

                    if (typeof value === 'string') {
                        // --- FILE ---
                        var filePath = value.replace('file:', '');
                        div.textContent = '📄 ' + key;
                        div.onmouseenter = function () { div.style.background = '#1a1a1a'; };
                        div.onmouseleave = function () { div.style.background = '';        };
                        div.onclick = function (event) {
                            event.stopPropagation();
                            console.log('Open file:', filePath);
                            window.currentFile = filePath;

                            fetch('/api/file-content?path=' + encodeURIComponent(filePath))
                                .then(function (res) { return res.text(); })
                                .then(function (fileData) {
                                    // *** CodeMirror-aware file open ***
                                    setEditorValue(fileData);

                                    // Push context to backend (non-blocking)
                                    fetch('/api/context/file', {
                                        method:  'POST',
                                        headers: { 'Content-Type': 'application/json' },
                                        body:    JSON.stringify({
                                            path: filePath, content: fileData, append_to_chat: false
                                        })
                                    }).catch(function (e) { console.warn('context/file:', e); });
                                })
                                .catch(function (err) { alert('Failed to load file: ' + err.message); });
                        };
                    } else {
                        // --- FOLDER ---
                        var folderDiv = document.createElement('div');
                        folderDiv.textContent = '📁 ' + key;
                        folderDiv.style.color = '#ff9d00';

                        var childrenDiv = document.createElement('div');
                        childrenDiv.style.display = 'none';

                        folderDiv.onclick = function (e) {
                            e.stopPropagation();
                            var open = childrenDiv.style.display !== 'none';
                            childrenDiv.style.display = open ? 'none' : 'block';
                            folderDiv.textContent = (open ? '📁 ' : '📂 ') + key;
                        };

                        renderTree(value, childrenDiv, indentLevel + 1);
                        div.appendChild(folderDiv);
                        div.appendChild(childrenDiv);
                        parentElement.appendChild(div);
                        return; // already appended
                    }
                    parentElement.appendChild(div);
                });
            }

            renderTree(tree, explorerDiv, 0);
        })
        .catch(function (error) {
            console.error('Explorer fetch error:', error);
            explorerDiv.innerHTML = 'Error loading files. Check F12 console.';
        });
};

// Kick off on first load
document.addEventListener('DOMContentLoaded', function () {
    window.loadExplorer();
});

async function runTerminalCommand() {
    var input  = document.getElementById('terminal-input');
    var output = document.getElementById('terminal-output');
    var cmd    = input.value.trim();
    if (!cmd) return;

    output.innerHTML += "<div><span style='color:#ff9d00;'>😊 " + escapeHtml(cmd) + "</span></div>";
    output.scrollTop = output.scrollHeight;
    input.value = '';

    try {
        var eventSource = new EventSource('/run_stream?cmd=' + encodeURIComponent(cmd));
        eventSource.onmessage = function (e) {
            var data = JSON.parse(e.data);
            if (data.line) {
                output.innerHTML += "<div style='color:#e0e0e0;margin:0;font-family:monospace;'>" + escapeHtml(data.line) + "</div>";
            } else if (data.done) {
                output.innerHTML += "<div style='color:#00ff00;font-weight:bold;'>Done ❤️</div>";
                eventSource.close();
            } else if (data.error) {
                output.innerHTML += "<div style='color:#ff0000;'>" + escapeHtml(data.error) + "</div>";
                eventSource.close();
            }
            output.scrollTop = output.scrollHeight;
        };
        eventSource.onerror = function () {
            output.innerHTML += "<div style='color:#00ff00;font-weight:bold;'>Done ❤️</div>";
            eventSource.close();
        };
    } catch (e) {
        output.innerHTML += "<span style='color:#ff0000;'>Failed: " + escapeHtml(e.message) + "</span>";
    }
}

function _afixRender() {
    var DiffLib = window.Diff || window.jsdiff;

    var orig  = window._autofix.original || '';

    if (!DiffLib || !DiffLib.diffLines) {
        leftEl.innerText  = orig;
        rightEl.innerText = fixed;
        return;
    }
    var parts = DiffLib.diffLines(orig, fixed);
    var leftHtml = '', rightHtml = '';
    parts.forEach(function (p) {
        var lines = p.value.split('\n');
        if (lines.length && lines[lines.length - 1] === '') lines.pop();
        lines.forEach(function (ln) {
            var safe = _afixEscape(ln) + '\n';
            if      (p.added)   { rightHtml += '<span class="ln ln-add">' + safe + '</span>'; }
            else if (p.removed) { leftHtml  += '<span class="ln ln-del">' + safe + '</span>'; }

                                  rightHtml += '<span class="ln ln-ctx">' + safe + '</span>'; }
        });
    });
    leftEl.innerHTML  = leftHtml;
    rightEl.innerHTML = rightHtml;

# Capture Guide

## Overview

VideoDB Capture enables real-time screen and audio recording with AI processing. Desktop capture currently supports **macOS** only.

For code-level details (SDK methods, event structures, AI pipelines), see [capture-reference.md](capture-reference.md).

## Quick Start

1. **Start WebSocket listener**: `python scripts/ws_listener.py --clear &`
2. **Run capture code** (see Complete Capture Workflow below)
3. **Events written to**: `/tmp/videodb_events.jsonl`

---

## Complete Capture Workflow

No webhooks or polling required. WebSocket delivers all events including session lifecycle.

> **CRITICAL:** The `CaptureClient` must remain running for the entire duration of the capture. It runs the local recorder binary that streams screen/audio data to VideoDB. If the Python process that created the `CaptureClient` exits, the recorder binary is killed and capture stops silently. Always run the capture code as a **long-lived background process** (e.g. `nohup python capture_script.py &`) and use signal handling (`asyncio.Event` + `SIGINT`/`SIGTERM`) to keep it alive until you explicitly stop it.

1. **Start WebSocket listener** in background with `--clear` flag to clear old events. Wait for it to create the WebSocket ID file.

2. **Read the WebSocket ID**. This ID is required for capture session and AI pipelines.

3. **Create a capture session** and generate a client token for the desktop client.

4. **Initialize CaptureClient** with the token. Request permissions for microphone and screen capture.

5. **List and select channels** (mic, display, system_audio). Set `store = True` on channels you want to persist as a video.

6. **Start the session** with selected channels.

7. **Wait for session active** by reading events until you see `capture_session.active`. This event contains the `rtstreams` array. Save session info (session ID, RTStream IDs) to a file (e.g. `/tmp/videodb_capture_info.json`) so other scripts can read it.

8. **Keep the process alive.** Use `asyncio.Event` with signal handlers for `SIGINT`/`SIGTERM` to block until explicitly stopped. Write a PID file (e.g. `/tmp/videodb_capture_pid`) so the process can be stopped later with `kill $(cat /tmp/videodb_capture_pid)`. The PID file should be overwritten on every run so reruns always have the correct PID.

9. **Start AI pipelines** (in a separate command/script) on each RTStream for audio indexing and visual indexing. Read the RTStream IDs from the saved session info file.

10. **Write custom event processing logic** (in a separate command/script) to read real-time events based on your use case. Examples:
    - Log Slack activity when `visual_index` mentions "Slack"
    - Summarize discussions when `audio_index` events arrive
    - Trigger alerts when specific keywords appear in `transcript`
    - Track application usage from screen descriptions

11. **Stop capture** when done — send SIGTERM to the capture process. It should call `client.stop_capture()` and `client.shutdown()` in its signal handler.

12. **Wait for export** by reading events until you see `capture_session.exported`. This event contains `exported_video_id`, `stream_url`, and `player_url`. This may take several seconds after stopping capture.

13. **Stop WebSocket listener** after receiving the export event. Use `kill $(cat /tmp/videodb_ws_pid)` to cleanly terminate it.

---

## Shutdown Sequence

Proper shutdown order is important to ensure all events are captured:

1. **Stop the capture session** — `client.stop_capture()` then `client.shutdown()`
2. **Wait for export event** — poll `/tmp/videodb_events.jsonl` for `capture_session.exported`
3. **Stop the WebSocket listener** — `kill $(cat /tmp/videodb_ws_pid)`

Do NOT kill the WebSocket listener before receiving the export event, or you will miss the final video URLs.
